    real = lib.Enum.zoneIDs

    lua.execute('local imp = QuestieLoader.ImportModule; local zdb = setmetatable({zoneIDs=QuestieDB.zoneIDs}, {__index=function(t,k) return imp(QuestieLoader, "ZoneDB_x")[k] end}); function QuestieLoader:ImportModule(n) if n == "ZoneDB" then return zdb end return imp(self, n) end')
